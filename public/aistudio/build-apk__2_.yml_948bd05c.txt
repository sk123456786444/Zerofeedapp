          # ZeroFeed icon: dark rounded square + white play triangle (Pillow)
          import os
          from PIL import Image, ImageDraw

          RES = "android/app/src/main/res"
          DENS = {"mdpi": (48, 108), "hdpi": (72, 162), "xhdpi": (96, 216),
                  "xxhdpi": (144, 324), "xxxhdpi": (192, 432)}
          BG = (10, 10, 10, 255)

          def triangle(d, size, scale):
              c = size / 2
              r = size * scale
              d.polygon([(c - r * 0.55, c - r), (c - r * 0.55, c + r), (c + r * 0.95, c)], fill=(255, 255, 255, 255))

          for name, (legacy, adaptive) in DENS.items():
              folder = f"{RES}/mipmap-{name}"
              if not os.path.isdir(folder): continue
              im = Image.new("RGBA", (legacy, legacy), (0, 0, 0, 0))
              d = ImageDraw.Draw(im)
              d.rounded_rectangle([0, 0, legacy - 1, legacy - 1], radius=legacy // 5, fill=BG)
              triangle(d, legacy, 0.26)
              im.save(f"{folder}/ic_launcher.png"); im.save(f"{folder}/ic_launcher_round.png")
              fg = Image.new("RGBA", (adaptive, adaptive), (0, 0, 0, 0))
              triangle(ImageDraw.Draw(fg), adaptive, 0.17)
              fg.save(f"{folder}/ic_launcher_foreground.png")

          bg_xml = f"{RES}/values/ic_launcher_background.xml"
          if os.path.isdir(f"{RES}/values"):
              open(bg_xml, "w").write('<?xml version="1.0" encoding="utf-8"?>\n<resources>\n'
                                      '    <color name="ic_launcher_background">#0A0A0A</color>\n</resources>\n')
          print("icon done")

          async function resolveDirect(item, force) {

            const hash = hashFromUrl(item.url);
            if (!hash) throw new Error('share link nahi');
            if (!force && directCache[hash]) return directCache[hash];
            const api = JD + '/sapi/media/set?action=get&items=true&thumbnaildimensions=true&shared=true&key=' + encodeURIComponent(hash);
            const r = await fetch(api, {
              method: 'POST',
              headers: { 'Content-Type': 'application/json', 'Accept': '*/*', 'Referer': JD + '/share/' + hash, 'Origin': JD },
              body: JSON.stringify({ data: {} })
            });

            const txt = await r.text();
            let data; try { data = JSON.parse(txt); } catch (e) { throw new Error('API jawab samajh nahi aaya'); }
            const url = findVideoUrl(data);
            if (!url) throw new Error('video link jawab mein nahi mila');
            const res = { url, poster: findPoster(data) };
            directCache[hash] = res;
            return res;
          }

          }

          function downloadItem(item) {

            resolveDirect(item).then((res) => { openExternal(res.url); toast('Browser mein download shuru hoga'); })
              .catch((e) => toast('Download nahi ho saka: ' + (e && e.message)));
          }
          function toggleFav(item, btn) {
            if (fav.has(item.url)) { fav.delete(item.url); btn.classList.remove('on'); toast('Favorites se hata diya'); }
            else { fav.add(item.url); btn.classList.add('on'); toast('Favorites mein add'); }
            persistSets(); buildChips();
          }
          function hideItem(s) {
            const i = slides.indexOf(s);
            hidden.add(s._item.url); persistSets();
            toast('Video chhupa di');
            render({ atIndex: i });
          }
          function videoMenu(s) {
            const item = s._item;
            showSheet((item.author ? '@' + item.author + ' — ' : '') + item.title.slice(0, 60), [
              { label: 'Download (browser mein)', fn: () => downloadItem(item) },
              { label: 'JazzDrive mein kholo', fn: () => openItem(item) },
              { label: 'Link copy karo', fn: () => copyText(item.url) },
              { label: 'Is video ko chhupao', danger: true, fn: () => hideItem(s) }

          /* slide ka andar ka hissa tab banta hai jab wo screen ke qareeb ho (hazaron videos par bhi halka) */
          function hydrate(s) {
            if (s._hyd || s._dead) return;
            s._hyd = true;
            const item = s._item;
            const bg = document.createElement('div'); bg.className = 'bg';
            if (item.thumb) bg.style.backgroundImage = 'url("' + item.thumb.replace(/"/g, '%22') + '")';
            else bg.style.background = gradientFor(item.title);
            s.appendChild(bg);

            const note = document.createElement('div'); note.className = 'note'; s.appendChild(note);
            const flash = document.createElement('div'); flash.className = 'flash'; s.appendChild(flash);
            const prog = document.createElement('div'); prog.className = 'prog'; s.appendChild(prog);

            const ov = document.createElement('div'); ov.className = 'overlay';
            const b = document.createElement('span'); b.className = 'badge'; b.textContent = item.platform;
            ov.appendChild(b);
            if (item.author) { const a = document.createElement('div'); a.className = 'author'; a.textContent = '@' + item.author; ov.appendChild(a); }
            const t = document.createElement('div'); t.className = 'title'; t.textContent = item.title;
            const m = document.createElement('div'); m.className = 'meta'; m.textContent = timeAgo(item.added);
            ov.append(t, m); s.appendChild(ov);

            const side = document.createElement('div'); side.className = 'side';
            const mk = (html, title, fn) => { const x = document.createElement('button'); x.className = 'sb'; x.innerHTML = html; x.title = title;
              x.onclick = (e) => { e.stopPropagation(); fn(x); }; side.appendChild(x); return x; };
            const snd = mk('', 'Awaz', () => {
              soundOn = !soundOn; store.set(K.sound, soundOn); updateSoundButtons();
              slides.forEach((x) => { if (x._v) x._v.muted = !soundOn; });
            });
            snd.classList.add('snd');
            mk('&#11015;', 'Download', () => downloadItem(item));
            const fb = mk('&#11088;', 'Favorite', (x) => toggleFav(item, x)); if (fav.has(item.url)) fb.classList.add('on');
            mk('&#8943;', 'Aur', () => videoMenu(s));
            s.appendChild(side);

            const fails = document.createElement('div'); fails.className = 'fails';
            const retry = document.createElement('button'); retry.className = 'fb'; retry.textContent = 'Dobara koshish';
            retry.onclick = (e) => { e.stopPropagation(); s._retried = false; activate(s, true); };
            const open = document.createElement('button'); open.className = 'fb'; open.textContent = 'JazzDrive mein kholo';
            open.onclick = (e) => { e.stopPropagation(); openItem(item); };
            fails.append(retry, open); s.appendChild(fails);

            if (seen.has(item.url)) addSeenBadge(s);
            s._el = { note, flash, prog };

              if (!v || !s._ready || s.classList.contains('failed')) return;
              if (v.paused) { pauseOthers(v); v.play().catch(() => {}); flash.textContent = '\u25B6'; } else { v.pause(); flash.textContent = '\u275A\u275A'; }
              flash.classList.add('show'); setTimeout(() => flash.classList.remove('show'), 450);
            });
          }
          function dehydrate(s) {
            if (!s._hyd) return;
            releaseSlide(s);
            s.textContent = ''; s._hyd = false; s._el = null; s.classList.remove('failed');
          }

          function buildChips() {
            const el = $('chips'); el.innerHTML = '';
            const base = items.filter((x) => !hidden.has(x.url));
            if (!base.length) { el.style.display = 'none'; return; }
            const add = (label, t, v) => {
              const b = document.createElement('button'); b.className = 'chip' + (filter.t === t && filter.v === v ? ' on' : ''); b.textContent = label;
              b.onclick = () => { filter = { t, v }; store.set(K.filter, filter); render({ atIndex: 0 }); }; el.appendChild(b);
            };
            add('Sab (' + base.length + ')', 'all', '');
            add('\u2B50 ' + base.filter((x) => fav.has(x.url)).length, 'fav', '');
            const plats = countBy(base, (x) => x.platform.toLowerCase());
            if (plats.length > 1) plats.forEach(([p, n]) => add(p.charAt(0).toUpperCase() + p.slice(1) + ' (' + n + ')', 'platform', p));
            const auth = countBy(base, (x) => x.author);
            if (auth.length > 1) auth.slice(0, 15).forEach(([a, n]) => add('@' + a + ' (' + n + ')', 'author', a));
            el.style.display = 'flex';
          }

          function currentUrl() { const s = slides.find((x) => x._vis); return s ? s._item.url : ''; }

          function render(opts) {
            opts = opts || {};
            const feed = $('feed');
            if (visObs) visObs.disconnect();
            if (nearObs) nearObs.disconnect();
            slides.forEach((s) => { s._dead = true; s._vis = false; releaseSlide(s); });   // purani slides ka kaam band
            document.querySelectorAll('video.v').forEach((v) => v.pause());
            feed.innerHTML = ''; slides = [];
            buildChips();
            const list = applyFilters();
            $('hideBtn').classList.toggle('on', hideSeen);
            $('nextBtn').classList.toggle('on', autoNext);
            $('empty').style.display = list.length ? 'none' : 'flex';
            if (!list.length) {
              $('empty').textContent = !items.length ? 'Abhi feed khali hai.\n\n' + lastDiag.join('\n') + '\n\nTelegram bot ko link bhejo, phir upar wala refresh dabao.'
                : (filter.t !== 'all' ? 'Is filter mein koi video nahi.\nUpar se "Sab" chuno.' : 'Sab videos dekh li!\nUpar wale \uD83D\uDC41 button se dobara dikhao.');
              $('count').textContent = '';
              return;
            }
            visObs = new IntersectionObserver((entries) => {
              entries.forEach((en) => {
                const s = en.target;
                if (s._dead) return;
                if (en.isIntersecting && en.intersectionRatio > 0.6) {
                  s._vis = true;
                  $('count').textContent = (Number(s.dataset.i) + 1) + ' / ' + list.length;
                  activate(s, false);
                } else {
                  s._vis = false;
                  if (s._v) { if (!en.isIntersecting || en.intersectionRatio < 0.1) releaseSlide(s); else s._v.pause(); }
                }
              });
            }, { threshold: [0, 0.1, 0.6, 1] });
            nearObs = new IntersectionObserver((entries) => {
              entries.forEach((en) => { const s = en.target; if (s._dead) return; if (en.isIntersecting) hydrate(s); else dehydrate(s); });
            }, { rootMargin: '150% 0px', threshold: 0 });

            list.forEach((item, i) => {
              const s = document.createElement('section'); s.className = 'slide'; s.dataset.i = i; s._item = item;
              slides.push(s); feed.appendChild(s); visObs.observe(s); nearObs.observe(s);
            });
            $('count').textContent = '1 / ' + list.length;

            let target = -1;
            if (opts.afterUrl) { const idx = list.findIndex((x) => x.url === opts.afterUrl); if (idx >= 0) target = Math.min(idx + 1, list.length - 1); }
            else if (opts.atUrl) target = list.findIndex((x) => x.url === opts.atUrl);
            else if (typeof opts.atIndex === 'number') target = Math.min(opts.atIndex, list.length - 1);
            if (target > 0) requestAnimationFrame(() => feed.scrollTo({ top: target * feed.clientHeight, behavior: 'instant' }));
          }

          function applyFeed(list, opts) {
            const sig = list.map((x) => x.url).join('|');
            if (sig === lastSig && slides.length && !(opts && opts.force)) return false;   // kuch naya nahi: video disturb na karo
            items = list; lastSig = sig;
            render(opts);
            return true;
          }

          async function refresh(silent) {
            if (!silent) toast('Feed refresh ho rahi hai...');
            const cur = currentUrl();
            const { list, live } = await loadFeed();
            const changed = applyFeed(list, { atUrl: cur, force: !silent });
            if (!silent) toast(live ? list.length + ' videos' : 'Feed load nahi hui (' + lastDiag.slice(1).join(', ') + ')');
            else if (changed && live) toast('Nayi videos aa gayi');
          }

          $('refBtn').onclick = () => refresh(false);

          $('nextBtn').onclick = () => {
            autoNext = !autoNext; store.set(K.next, autoNext);

            slides.forEach((s) => { if (s._v) s._v.loop = !autoNext; });
            toast(autoNext ? 'Auto-next on: video khatam to agli' : 'Loop on: video dobara chalegi');
          };
          function askUrl() {
            const u = prompt('Feed.json ka URL (khali chhodo = default):', store.get(K.url, ''));
            if (u !== null) { if (u.trim()) store.set(K.url, u.trim()); else store.del(K.url); refresh(false); }
          }
          $('cfgBtn').onclick = () => showSheet('ZeroFeed ' + VERSION + ' — settings', [
            { label: 'Feed ka URL badlo', fn: askUrl },

            { label: 'Chhupayi hui videos wapas lao (' + hidden.size + ')', fn: () => { hidden.clear(); persistSets(); render({ atIndex: 0 }); toast('Wapas aa gayi'); } },
            { label: 'Cache saaf karke feed dobara lo', fn: () => { store.del(K.feed); lastSig = ''; refresh(false); } }
          ]);
          let pressTimer = null;
          $('brand').addEventListener('touchstart', () => { pressTimer = setTimeout(askUrl, 1000); });
          ['touchend', 'touchmove'].forEach((ev) => $('brand').addEventListener(ev, () => clearTimeout(pressTimer)));

          /* app background mein jaye to awaz band; wapas aaye to chalo + feed check */
          document.addEventListener('visibilitychange', () => {
            if (document.hidden) { pauseOthers(null); return; }
            const cur = slides.find((s) => s._vis); if (cur) playVideo(cur);
            if (Date.now() - lastRefresh > 180000) refresh(true);
          });

          (async () => {
            const cached = normalize(store.get(K.feed, []));
            const back = store.get(K.ret, null); store.del(K.ret);
            const resume = store.get(K.last, '');
            const startOpts = back ? { afterUrl: back } : { atUrl: resume };
            if (cached.length) applyFeed(cached, startOpts);
            const { list, live } = await loadFeed();
            if (live) applyFeed(list, cached.length ? { atUrl: currentUrl() || resume } : startOpts);
            else if (!cached.length) { applyFeed(list, startOpts); toast('Feed load nahi hui — internet check karo'); }
          })();
